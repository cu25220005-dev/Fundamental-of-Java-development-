// Q3. Count Characters, Words and Lines

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class Lab10_Q03 {
    public static void main(String[] args) {
        System.out.println("--- Q3: Count Characters, Words and Lines ---");
        int lines = 0, words = 0, chars = 0;
        try {
            BufferedReader reader = new BufferedReader(new FileReader("data.txt"));
            String line;
            while ((line = reader.readLine()) != null) {
                lines++;
                chars += line.length();
                String[] wordList = line.trim().split("\s+");
                if (!line.trim().isEmpty()) {
                    words += wordList.length;
                }
            }
            reader.close();
            System.out.println("Total Lines: " + lines);
            System.out.println("Total Words: " + words);
            System.out.println("Total Characters: " + chars);
        } catch (IOException e) {
            System.out.println("Error reading 'data.txt': " + e.getMessage());
        }
    }
}

// Q6. Search for a Word in a File

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;
import java.util.Scanner;

public class Lab10_Q06 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q6: Search for a Word in a File ---");
        System.out.print("Enter word to search in 'notes.txt': ");
        String targetWord = sc.next();
        int count = 0;

        try {
            BufferedReader reader = new BufferedReader(new FileReader("notes.txt"));
            String line;
            while ((line = reader.readLine()) != null) {
                String[] words = line.split("\s+");
                for (String w : words) {
                    if (w.equalsIgnoreCase(targetWord)) {
                        count++;
                    }
                }
            }
            reader.close();
            if (count > 0) {
                System.out.println("Word '" + targetWord + "' found! Occurrences: " + count);
            } else {
                System.out.println("Word '" + targetWord + "' not found in the file.");
            }
        } catch (IOException e) {
            System.out.println("Error reading 'notes.txt': " + e.getMessage());
        }
        sc.close();
    }
}

// Q8. Count Vowels, Consonants, Digits and Spaces

import java.io.FileReader;
import java.io.IOException;

public class Lab10_Q08 {
    public static void main(String[] args) {
        System.out.println("--- Q8: Count Vowels, Consonants, Digits and Spaces ---");
        int vowels = 0, consonants = 0, digits = 0, spaces = 0;
        try {
            FileReader reader = new FileReader("input.txt");
            int ch;
            while ((ch = reader.read()) != -1) {
                char c = Character.toLowerCase((char) ch);
                if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u') {
                    vowels++;
                } else if (c >= 'a' && c <= 'z') {
                    consonants++;
                } else if (Character.isDigit(c)) {
                    digits++;
                } else if (Character.isWhitespace(c)) {
                    spaces++;
                }
            }
            reader.close();
            System.out.println("Vowels: " + vowels);
            System.out.println("Consonants: " + consonants);
            System.out.println("Digits: " + digits);
            System.out.println("Spaces: " + spaces);
        } catch (IOException e) {
            System.out.println("Error reading 'input.txt': " + e.getMessage());
        }
    }
}

            BufferedReader reader = new BufferedReader(new FileReader("student_records.txt"));

// Q10. File-Based Employee Management

import java.io.FileWriter;
import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;
import java.io.FileNotFoundException;
import java.util.Scanner;

public class Lab10_Q10 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q10: File-Based Employee Management ---");
        try {
            FileWriter writer = new FileWriter("employees.txt", true);
            System.out.print("Enter Employee ID: ");
            String id = sc.next();
            System.out.print("Enter Employee Name: ");
            String name = sc.next();
            System.out.print("Enter Department: ");
            String dept = sc.next();
            System.out.print("Enter Salary: ");
            double salary = sc.nextDouble();

            writer.write(id + "," + name + "," + dept + "," + salary + "\n");
            writer.close();
            System.out.println("Employee record stored successfully.\n");

            System.out.print("Enter Employee ID to search: ");
            String searchId = sc.next();
            boolean found = false;

            BufferedReader reader = new BufferedReader(new FileReader("employees.txt"));
            String line;
            while ((line = reader.readLine()) != null) {
                String[] data = line.split(",");
                if (data[0].equals(searchId)) {
                    System.out.println("Employee Record Found:");
                    System.out.println("ID: " + data[0] + " | Name: " + data[1] + " | Dept: " + data[2] + " | Salary: ₹" + data[3]);
                    found = true;
                    break;
                }
            }
            reader.close();

            if (!found) {
                System.out.println("Employee record not found.");
            }
        } catch (FileNotFoundException e) {
            System.out.println("Employee record not found.");
        } catch (IOException e) {
            System.out.println("An error occurred: " + e.getMessage());
        }
        sc.close();
    }
}
