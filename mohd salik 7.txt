// Q2. Bank Account with Validation

import java.util.Scanner;

class BankAccountEncapsulated {
    private String accountNumber;
    private String accountHolder;
    private double balance;

    public void setAccountNumber(String accNo) { this.accountNumber = accNo; }
    public String getAccountNumber() { return accountNumber; }

    public void setAccountHolder(String holder) { this.accountHolder = holder; }
    public String getAccountHolder() { return accountHolder; }

    public double getBalance() { return balance; }

    public void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            System.out.println("Deposited ₹" + amount + ". New Balance: ₹" + balance);
        } else {
            System.out.println("Invalid deposit amount! Must be greater than 0.");
        }
    }

    public void withdraw(double amount) {
        if (amount > 0 && amount <= balance) {
            balance -= amount;
            System.out.println("Withdrawn ₹" + amount + ". Remaining Balance: ₹" + balance);
        } else {
            System.out.println("Invalid withdrawal! Amount exceeds balance or is invalid.");
        }
    }
}

public class Lab07_Q02 {
    public static void main(String[] args) {
        System.out.println("--- Q2: Bank Account with Validation ---");
        BankAccountEncapsulated acc = new BankAccountEncapsulated();
        acc.setAccountNumber("ACC123");
        acc.setAccountHolder("Aarav");
        acc.deposit(5000);
        acc.withdraw(2000);
        acc.withdraw(4000);
    }
}



// Q3. Employee Salary Encapsulation with Validation

import java.util.Scanner;

class EmployeeSalary {
    private int employeeId;
    private String employeeName;
    private double salary;

    public void setEmployeeId(int id) { this.employeeId = id; }
    public int getEmployeeId() { return employeeId; }

    public void setEmployeeName(String name) { this.employeeName = name; }
    public String getEmployeeName() { return employeeName; }

    public void setSalary(double salary) {
        if (salary >= 0 && salary <= 1000000) {
            this.salary = salary;
            System.out.println("Salary set successfully: ₹" + salary);
        } else {
            System.out.println("Invalid salary (" + salary + ")! Must be between 0 and 1,000,000.");

    public double getSalary() { return salary; }

    public void display() {
        System.out.println("Emp ID: " + employeeId + ", Name: " + employeeName + ", Salary: ₹" + salary);
    }
}

// Q15. Shape Area Overriding

import java.util.Scanner;

class Shape {
    void calculateArea() { System.out.println("Area calculation not defined for generic shape."); }
}

class CircleQ15 extends Shape {
    double radius = 7.0;
    @Override
    void calculateArea() {
        double area = Math.PI * radius * radius;
        System.out.println("Circle Area (r=7): " + area);
    }
}

class RectangleQ15 extends Shape {
    double length = 10.0, breadth = 5.0;
    @Override
    void calculateArea() {
        double area = length * breadth;
        System.out.println("Rectangle Area (10x5): " + area);
    }
}

public class Lab07_Q15 {
    public static void main(String[] args) {
        System.out.println("--- Q15: Shape Area Overriding ---");
        Shape c = new CircleQ15(); c.calculateArea();
        Shape r = new RectangleQ15(); r.calculateArea();
    }
}

// Q20. University Employee System - Integrated OOP

import java.util.Scanner;

interface ResearcherQ20 {
    void conductResearch();
}

abstract class EmployeeQ20 {
    private int employeeId;
    private String employeeName;
    private double salary;

    public EmployeeQ20(int id, String name, double salary) {
        this.employeeId = id;
        this.employeeName = name;
        this.salary = salary;
    }

    public int getEmployeeId() { return employeeId; }
    public String getEmployeeName() { return employeeName; }
    public double getSalary() { return salary; }

    public void displayDetails() {
        System.out.println("ID: " + employeeId + ", Name: " + employeeName + ", Base Salary: ₹" + salary);
    }

    abstract void calculateSalary();
}

class TeacherQ20 extends EmployeeQ20 implements ResearcherQ20 {
    String subject;

    public TeacherQ20(int id, String name, double salary, String subject) {
        super(id, name, salary);
        this.subject = subject;
    }

    @Override
    void calculateSalary() {
        double total = getSalary() + 10000;
        System.out.println("Teacher (" + subject + ") Total Salary: ₹" + total);
    }

    @Override
    public void conductResearch() {
        System.out.println(getEmployeeName() + " is conducting academic research in " + subject);
    }
}

class VisitingTeacher extends EmployeeQ20 {
    int hoursWorked;
    double ratePerHour;

    public VisitingTeacher(int id, String name, int hours, double rate) {
        super(id, name, 0);
        this.hoursWorked = hours;
        this.ratePerHour = rate;
    }

    @Override
    void calculateSalary() {
        double total = hoursWorked * ratePerHour;
        System.out.println("Visiting Teacher Total Salary (" + hoursWorked + " hrs): ₹" + total);
    }
}

class Admin extends EmployeeQ20 {
    String department;

    public Admin(int id, String name, double salary, String department) {
        super(id, name, salary);
        this.department = department;
    }

    @Override
    void calculateSalary() {
        double total = getSalary() + 15000;
        System.out.println("Admin (" + department + ") Total Salary: ₹" + total);
    }

    void manageDepartment() {
        System.out.println(getEmployeeName() + " is managing the " + department + " department.");
    }
}

public class Lab07_Q20 {
    public static void main(String[] args) {
        System.out.println("--- Q20: University Employee System (Integrated OOP) ---");
        TeacherQ20 t = new TeacherQ20(901, "Dr. Sharma", 60000, "Computer Science");
        t.displayDetails(); t.calculateSalary(); t.conductResearch();

        System.out.println();
        VisitingTeacher vt = new VisitingTeacher(902, "Prof. Verma", 25, 1200);
        vt.displayDetails(); vt.calculateSalary();

        System.out.println();
        Admin adm = new Admin(903, "Mr. Kapoor", 50000, "Examinations");
        adm.displayDetails(); adm.calculateSalary(); adm.manageDepartment();
    }
}
