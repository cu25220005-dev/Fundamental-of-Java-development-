// Q9. ListIterator Modification

import java.util.List;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.ListIterator;

public class Lab09_Q09 {
    public static void main(String[] args) {
        System.out.println("--- Q9: ListIterator Modification ---");
        List<String> frameworkList = new ArrayList<>(Arrays.asList("Spring", "Hibernate", "Struts", "JSF"));
        System.out.println("Original List: " + frameworkList);

        ListIterator<String> litr = frameworkList.listIterator();
        while (litr.hasNext()) {
            String item = litr.next();
            if (item.equals("Struts")) {
                litr.set("Spring Boot"); // Replace element
            }
            if (item.equals("Hibernate")) {
                litr.add("Micronaut"); // Insert element during iteration
            }
        }
        System.out.println("Modified List using ListIterator: " + frameworkList);
    }
}

// Q13. ListIterator Report

import java.util.ArrayList;
import java.util.Arrays;
import java.util.ListIterator;

public class Lab09_Q13 {
    public static void main(String[] args) {
        System.out.println("--- Q13: ListIterator Report (Forward & Backward) ---");
        ArrayList<String> subjectList = new ArrayList<>(Arrays.asList("Java", "Python", "C++", "DSA", "OS"));
        ListIterator<String> subjectItr = subjectList.listIterator();
        
        System.out.println("Traversing Forward:");
        while (subjectItr.hasNext()) {
            System.out.println("Index: " + subjectItr.nextIndex() + ", Element: " + subjectItr.next());
        }
        
        System.out.println("Traversing Backward:");
        while (subjectItr.hasPrevious()) {
            System.out.println("Index: " + subjectItr.previousIndex() + ", Element: " + subjectItr.previous());
        }
    }
}



// Q14. ArrayList vs LinkedList Lab

import java.util.ArrayList;
import java.util.LinkedList;

public class Lab09_Q14 {
    public static void main(String[] args) {
        System.out.println("--- Q14: ArrayList vs LinkedList Lab ---");
        ArrayList<String> testArr = new ArrayList<>();
        LinkedList<String> testLink = new LinkedList<>();
        
        testArr.add("Item 1"); testArr.add("Item 2");
        testLink.add("Item 1"); testLink.add("Item 2");
        
        System.out.println("ArrayList get(0): " + testArr.get(0) + ", Size: " + testArr.size());
        System.out.println("LinkedList get(0): " + testLink.get(0) + ", Size: " + testLink.size());
        
        testArr.remove(0);
        testLink.remove(0);
        System.out.println("After removal - ArrayList size: " + testArr.size() + ", LinkedList size: " + testLink.size());
        System.out.println("Observation: ArrayList uses dynamic array (fast random access O(1), slow middle insertions O(n)). LinkedList uses doubly linked list (fast insertions O(1), slow random access O(n)).");
    }
}



// Q15. Mini Student Collection Management System

import java.util.ArrayList;
import java.util.HashSet;
import java.util.TreeSet;
import java.util.Iterator;

class MiniStudentSystem {
    private ArrayList<String> insertionOrderList = new ArrayList<>();
    private HashSet<String> uniqueIdSet = new HashSet<>();
    private TreeSet<String> sortedIdSet = new TreeSet<>();

    public void addStudent(String id, String name) {
        if (uniqueIdSet.add(id)) {
            sortedIdSet.add(id);
            insertionOrderList.add(id + " - " + name);
            System.out.println("Student added: " + name + " (ID: " + id + ")");
        } else {
            System.out.println("Error: Student ID " + id + " already exists!");
        }
    }

    public void removeStudent(String id) {
        boolean removed = false;
        Iterator<String> it = insertionOrderList.iterator();
        while (it.hasNext()) {
            String record = it.next();
            if (record.startsWith(id + " -")) {
                it.remove();
                removed = true;
                break;
            }
        }
        if (removed) {
            uniqueIdSet.remove(id);
            sortedIdSet.remove(id);
            System.out.println("Student ID " + id + " removed.");
        } else {
            System.out.println("Student ID " + id + " not found.");
        }
    }

    public void searchStudent(String id) {
        if (uniqueIdSet.contains(id)) {
            System.out.println("Student ID " + id + " found in system.");
        } else {
            System.out.println("Student ID " + id + " not found.");
        }
    }

    public void displayAll() {
        System.out.println("\n--- Student Records (ArrayList - Insertion Order) ---");
        for (String s : insertionOrderList) System.out.println(s);
        System.out.println("Total Count: " + insertionOrderList.size());
        
        System.out.println("\n--- Unique Student IDs (HashSet) ---");
        System.out.println(uniqueIdSet);

        System.out.println("\n--- Sorted Student IDs (TreeSet) ---");
        System.out.println(sortedIdSet);
    }
}

public class Lab09_Q15 {
    public static void main(String[] args) {
        System.out.println("--- Q15: Mini Student Collection Management System ---");
        MiniStudentSystem system = new MiniStudentSystem();
        system.addStudent("S105", "Aarav");
        system.addStudent("S101", "Priya");
        system.addStudent("S103", "Rahul");
        system.addStudent("S101", "Duplicate ID Test");
        
        system.displayAll();
        
        System.out.println("\nSearching for S103:");
        system.searchStudent("S103");
        
        System.out.println("\nRemoving S105:");
        system.removeStudent("S105");
        
        system.displayAll();
    }
}
