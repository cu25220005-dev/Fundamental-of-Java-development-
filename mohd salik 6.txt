// Q1. Handling ArithmeticException

import java.util.Scanner;

public class Lab06_Q01 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q1: Handling ArithmeticException ---");
        try {
            System.out.print("Enter numerator: ");
            int a = sc.nextInt();
            System.out.print("Enter denominator: ");
            int b = sc.nextInt();
            int result = a / b;
            System.out.println("Result: " + result);
        } catch (ArithmeticException e) {
            System.out.println("Caught Exception: Division by zero is not allowed (" + e.getMessage() + ")");
        }
        sc.close();
    }
}

// Q4. Multiple catch Blocks

import java.util.Scanner;

public class Lab06_Q04 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q4: Multiple catch Blocks Demo ---");
        int[] arr = {5, 10, 15, 20, 25};
        try {
            System.out.print("Enter numerator for division: ");
            int a = Integer.parseInt(sc.next());
            System.out.print("Enter denominator for division: ");
            int b = Integer.parseInt(sc.next());
            System.out.println("Division Result: " + (a / b));

            System.out.print("Enter array index to access (0-4): ");
            int idx = sc.nextInt();
            System.out.println("Array Element: " + arr[idx]);
        } catch (ArithmeticException e) {
            System.out.println("Caught ArithmeticException: Division by zero.");
        } catch (ArrayIndexOutOfBoundsException e) {
            System.out.println("Caught ArrayIndexOutOfBoundsException: Index out of bounds.");
        } catch (NumberFormatException e) {
            System.out.println("Caught NumberFormatException: Invalid number format.");
        }
        sc.close();
    }
}

// Q8. Nested try-catch

import java.util.Scanner;

public class Lab06_Q08 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q8: Nested try-catch ---");
        int[] arr = {1, 2, 3};
        try {
            // Outer try block for division
            System.out.print("Enter numerator: ");
            int a = sc.nextInt();
            System.out.print("Enter denominator: ");
            int b = sc.nextInt();
            System.out.println("Division Result: " + (a / b));

            try {
                // Inner try block for array access
                System.out.print("Enter array index (0-2): ");
                int idx = sc.nextInt();
                System.out.println("Array Element: " + arr[idx]);
            } catch (ArrayIndexOutOfBoundsException e) {
                System.out.println("Inner Catch: Invalid array index (" + e.getMessage() + ")");
            }

        } catch (ArithmeticException e) {
            System.out.println("Outer Catch: Division by zero (" + e.getMessage() + ")");
        }
        sc.close();
    }
}



// Q9. Exception Propagation

import java.util.Scanner;

public class Lab06_Q09 {
    public static void method3() {
        int x = 10 / 0; // Generates ArithmeticException
    }

    public static void method2() {
        method3();
    }

    public static void method1() {
        method2();
    }

    public static void main(String[] args) {
        System.out.println("--- Q9: Exception Propagation ---");
        try {
            method1();
        } catch (ArithmeticException e) {
            System.out.println("Caught propagated exception in main: " + e.getMessage());
        }
    }
}

// Q11. Bank Withdrawal System

import java.util.Scanner;

class InsufficientBalanceException extends Exception {
    public InsufficientBalanceException(String msg) { super(msg); }
}

public class Lab06_Q11 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q11: Bank Withdrawal System ---");
        try {
            System.out.print("Enter account balance: ");
            double balance = Double.parseDouble(sc.next());
            System.out.print("Enter withdrawal amount: ");
            double amount = Double.parseDouble(sc.next());

            if (amount < 0) {
                System.out.println("Error: Withdrawal amount cannot be negative.");
            } else if (amount > balance) {
                throw new InsufficientBalanceException("Withdrawal amount exceeds available balance.");
            } else {
                balance -= amount;
                System.out.println("Withdrawal successful! Remaining balance: ₹" + balance);
            }
        } catch (NumberFormatException e) {
            System.out.println("Error: Invalid numeric input entered.");
        } catch (InsufficientBalanceException e) {
            System.out.println("Custom Exception Caught: " + e.getMessage());
        } finally {
            System.out.println("Bank transaction completed.");
        }
        sc.close();
    }
}

// Q15. Pharmacy Inventory System

import java.util.Scanner;

class InvalidQuantityException extends Exception {
    public InvalidQuantityException(String msg) { super(msg); }
}

class InsufficientMedicineStockException extends Exception {
    public InsufficientMedicineStockException(String msg) { super(msg); }
}

public class Lab06_Q15 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q15: Pharmacy Inventory System ---");
        try {
            System.out.print("Enter medicine name: ");
            String med = sc.next();
            System.out.print("Enter available quantity: ");
            int available = Integer.parseInt(sc.next());
            System.out.print("Enter required quantity: ");
            int required = Integer.parseInt(sc.next());

            if (available < 0 || required < 0) {
                throw new InvalidQuantityException("Quantities cannot be negative.");
            }
            if (required > available) {
                throw new InsufficientMedicineStockException("Required quantity exceeds available stock.");
            }
            System.out.println("Inventory transaction approved for " + med + ". Remaining stock: " + (available - required));
        } catch (NumberFormatException e) {
            System.out.println("Error: Invalid numeric input entered.");
        } catch (InvalidQuantityException e) {
            System.out.println("InvalidQuantityException Caught: " + e.getMessage());
        } catch (InsufficientMedicineStockException e) {
            System.out.println("InsufficientMedicineStockException Caught: " + e.getMessage());
        } finally {
            System.out.println("Inventory transaction completed.");
        }
        sc.close();
    }
}
