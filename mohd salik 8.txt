// Q1. Generic Box

import java.util.Scanner;

class Box<T> {
    private T value;

    public void set(T value) { this.value = value; }
    public T get() { return value; }
}

public class Lab08_Q01 {
    public static void main(String[] args) {
        System.out.println("--- Q1: Generic Box ---");
        Box<Integer> intBox = new Box<>();
        intBox.set(100);
        System.out.println("Integer Box Value: " + intBox.get());

        Box<String> strBox = new Box<>();
        strBox.set("COER University");
        System.out.println("String Box Value: " + strBox.get());

        Box<Double> dblBox = new Box<>();
        dblBox.set(99.99);
        System.out.println("Double Box Value: " + dblBox.get());
    }
}

// Q14. Generic Method and ArrayList

import java.util.ArrayList;

public class Lab08_Q14 {
    public static <T> void displayList(ArrayList<T> list) {
        System.out.print("List Contents: ");

        System.out.println();
    }

    public static void main(String[] args) {
        System.out.println("--- Q14: Generic Method with ArrayList ---");

        displayList(intList);
        displayList(strList);
    }
}



// Q15. Mini Generic Student Management Program

import java.util.ArrayList;
import java.util.Scanner;

class ManagementStudent<T, U, V> {
    private T id;
    private U name;
    private V marks;

    public ManagementStudent(T id, U name, V marks) {
        this.id = id;
        this.name = name;
        this.marks = marks;
    }

    public T getId() { return id; }
    public U getName() { return name; }
    public V getMarks() { return marks; }

    public void display() {
        System.out.println("Student ID: " + id + " | Name: " + name + " | Marks: " + marks);
    }
}

public class Lab08_Q15 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q15: Mini Generic Student Management Program ---");
        ArrayList<ManagementStudent<Integer, String, Double>> studentList = new ArrayList<>();

        studentList.add(new ManagementStudent<>(201, "Aarav", 85.5));
        studentList.add(new ManagementStudent<>(202, "Priya", 91.0));
        studentList.add(new ManagementStudent<>(203, "Rahul", 78.0));
        studentList.add(new ManagementStudent<>(204, "Sneha", 88.5));
        studentList.add(new ManagementStudent<>(205, "Amit", 79.0));

        System.out.println("\nAll Students:");
        for (ManagementStudent<Integer, String, Double> s : studentList) {
            s.display();
        }

        System.out.println("\nTotal number of students (size): " + studentList.size());

        System.out.print("\nEnter student ID to search: ");
        int searchId = sc.nextInt();
        boolean found = false;
        for (ManagementStudent<Integer, String, Double> s : studentList) {
            if (s.getId().equals(searchId)) {
                System.out.println("Student found!");
                s.display();
                found = true;
                break;
            }
        }
        if (!found) {
            System.out.println("Student with ID " + searchId + " not found.");
        }
        sc.close();
    }
}
