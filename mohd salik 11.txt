// Q2. Simple Calculator - AWT

import java.awt.*;
import java.awt.event.*;

public class Lab11_Q02 extends Frame implements ActionListener {
    TextField tf1, tf2;
    Button btnAdd, btnSub, btnMul, btnDiv;
    Label lblResult;

    public Lab11_Q02() {
        setTitle("Simple Calculator - AWT");
        setSize(300, 250);
        setLayout(new FlowLayout());

        add(new Label("Number 1:"));
        tf1 = new TextField(15);
        add(tf1);

        add(new Label("Number 2:"));
        tf2 = new TextField(15);
        add(tf2);

        btnAdd = new Button("+");
        btnSub = new Button("-");
        btnMul = new Button("*");
        btnDiv = new Button("/");
        add(btnAdd); add(btnSub); add(btnMul); add(btnDiv);

        lblResult = new Label("Result: ");
        add(lblResult);

        btnAdd.addActionListener(this);
        btnSub.addActionListener(this);
        btnMul.addActionListener(this);
        btnDiv.addActionListener(this);

        addWindowListener(new WindowAdapter() {
            public void windowClosing(WindowEvent we) { dispose(); }
        });
    }

    public void actionPerformed(ActionEvent e) {
        try {
            double n1 = Double.parseDouble(tf1.getText());
            double n2 = Double.parseDouble(tf2.getText());
            double res = 0;

            if (e.getSource() == btnAdd) res = n1 + n2;
            else if (e.getSource() == btnSub) res = n1 - n2;
            else if (e.getSource() == btnMul) res = n1 * n2;
            else if (e.getSource() == btnDiv) {
                if (n2 == 0) {
                    lblResult.setText("Error: Division by zero!");
                    return;
                }
                res = n1 / n2;
            }
            lblResult.setText("Result: " + res);
        } catch (NumberFormatException ex) {
            lblResult.setText("Error: Enter valid numbers.");
        }
    }

    public static void main(String[] args) {
        System.out.println("--- Q2: Simple Calculator AWT ---");
        new Lab11_Q02().setVisible(true);
    }
}

import javax.swing.*;

public class Lab11_Q07 extends JFrame implements ActionListener {
    JTextField tfTemp;
    JRadioButton rbC2F, rbF2C;
    ButtonGroup bg;
    JButton btnConvert, btnClear;
    JLabel lblResult;

    public Lab11_Q07() {
        setTitle("Temperature Converter - Swing");
        setSize(350, 250);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLayout(null);

        JLabel l1 = new JLabel("Temperature:"); l1.setBounds(30, 20, 90, 25); add(l1);
        tfTemp = new JTextField(); tfTemp.setBounds(130, 20, 160, 25); add(tfTemp);

        rbC2F = new JRadioButton("Celsius to Fahrenheit", true); rbC2F.setBounds(30, 60, 160, 25); add(rbC2F);
        rbF2C = new JRadioButton("Fahrenheit to Celsius"); rbF2C.setBounds(30, 90, 160, 25); add(rbF2C);
        bg = new ButtonGroup(); bg.add(rbC2F); bg.add(rbF2C);

        btnConvert = new JButton("Convert"); btnConvert.setBounds(50, 130, 90, 30); add(btnConvert);
        btnClear = new JButton("Clear"); btnClear.setBounds(170, 130, 90, 30); add(btnClear);

        lblResult = new JLabel("Result: "); lblResult.setBounds(30, 175, 280, 25); add(lblResult);

        btnConvert.addActionListener(this);
        btnClear.addActionListener(this);
    }

    public void actionPerformed(ActionEvent e) {
        if (e.getSource() == btnConvert) {
            try {
                double val = Double.parseDouble(tfTemp.getText());
                if (rbC2F.isSelected()) {
                    double f = (val * 9.0 / 5.0) + 32;
                    lblResult.setText("Result: " + val + " °C = " + f + " °F");
                } else {
                    double c = (val - 32) * 5.0 / 9.0;
                    lblResult.setText("Result: " + val + " °F = " + c + " °C");
                }
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(this, "Please enter a valid numeric temperature.", "Error", JOptionPane.ERROR_MESSAGE);
            }
        } else if (e.getSource() == btnClear) {
            tfTemp.setText("");
            rbC2F.setSelected(true);
            lblResult.setText("Result: ");
        }
    }

    public static void main(String[] args) {
        System.out.println("--- Q7: Temperature Converter Swing ---");
        new Lab11_Q07().setVisible(true);
    }
}



// Q8. Simple Notepad - Swing

import javax.swing.*;
import java.awt.event.*;
import java.io.*;

public class Lab11_Q08 extends JFrame implements ActionListener {
    JTextArea textArea;
    JMenuBar menuBar;
    JMenu fileMenu;
    JMenuItem itemNew, itemOpen, itemSave, itemExit;
    JFileChooser fileChooser;

    public Lab11_Q08() {
        setTitle("Simple Notepad - Swing");
        setSize(500, 400);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);

        textArea = new JTextArea();
        add(new JScrollPane(textArea), "Center");

        menuBar = new JMenuBar();
        fileMenu = new JMenu("File");

        itemNew = new JMenuItem("New");
        itemOpen = new JMenuItem("Open");
        itemSave = new JMenuItem("Save");
        itemExit = new JMenuItem("Exit");

        fileMenu.add(itemNew); fileMenu.add(itemOpen); fileMenu.add(itemSave); fileMenu.addSeparator(); fileMenu.add(itemExit);
        menuBar.add(fileMenu);
        setJMenuBar(menuBar);

        itemNew.addActionListener(this);
        itemOpen.addActionListener(this);
        itemSave.addActionListener(this);
        itemExit.addActionListener(this);

        fileChooser = new JFileChooser();
    }

    public void actionPerformed(ActionEvent e) {
        if (e.getSource() == itemNew) {
            textArea.setText("");
        } else if (e.getSource() == itemOpen) {
            if (fileChooser.showOpenDialog(this) == JFileChooser.APPROVE_OPTION) {
                File file = fileChooser.getSelectedFile();
                try (BufferedReader br = new BufferedReader(new FileReader(file))) {
                    textArea.read(br, null);
                } catch (IOException ex) {
                    JOptionPane.showMessageDialog(this, "Error opening file.", "Error", JOptionPane.ERROR_MESSAGE);
                }
            }
        } else if (e.getSource() == itemSave) {
            if (fileChooser.showSaveDialog(this) == JFileChooser.APPROVE_OPTION) {
                File file = fileChooser.getSelectedFile();
                try (BufferedWriter bw = new BufferedWriter(new FileWriter(file))) {
                    textArea.write(bw);
                } catch (IOException ex) {
                    JOptionPane.showMessageDialog(this, "Error saving file.", "Error", JOptionPane.ERROR_MESSAGE);
                }
            }
        } else if (e.getSource() == itemExit) {
            System.exit(0);
        }
    }

    public static void main(String[] args) {
        System.out.println("--- Q8: Simple Notepad Swing ---");
        new Lab11_Q08().setVisible(true);
    }
}

        JLabel l2 = new JLabel("Customer Name:"); l2.setBounds(30, 65, 120, 25); add(l2);

        JLabel l3 = new JLabel("Current Balance:"); l3.setBounds(30, 105, 120, 25); add(l3);
        tfBalance = new JTextField(String.valueOf(balance)); tfBalance.setBounds(160, 105, 180, 25); tfBalance.setEditable(false); add(tfBalance);

        JLabel l4 = new JLabel("Amount:"); l4.setBounds(30, 145, 120, 25); add(l4);
        tfAmount = new JTextField(); tfAmount.setBounds(160, 145, 180, 25); add(tfAmount);

        btnDeposit = new JButton("Deposit"); btnDeposit.setBounds(40, 200, 95, 30); add(btnDeposit);
        btnWithdraw = new JButton("Withdraw"); btnWithdraw.setBounds(145, 200, 95, 30); add(btnWithdraw);
        btnCheck = new JButton("Check Bal"); btnCheck.setBounds(250, 200, 95, 30); add(btnCheck);

        btnClear = new JButton("Clear"); btnClear.setBounds(145, 245, 95, 30); add(btnClear);

        btnDeposit.addActionListener(this);
        btnWithdraw.addActionListener(this);
        btnCheck.addActionListener(this);
        btnClear.addActionListener(this);
    }

    public void actionPerformed(ActionEvent e) {
        try {
            if (e.getSource() == btnCheck) {
                JOptionPane.showMessageDialog(this, "Current Balance: ₹" + balance, "Balance Inquiry", JOptionPane.INFORMATION_MESSAGE);
                return;
            }
            if (e.getSource() == btnClear) {
                tfAmount.setText("");
                return;
            }

            double amt = Double.parseDouble(tfAmount.getText());
            if (amt <= 0) {
                JOptionPane.showMessageDialog(this, "Please enter a valid positive amount.", "Error", JOptionPane.ERROR_MESSAGE);
                return;
            }

            if (e.getSource() == btnDeposit) {
                balance += amt;
                tfBalance.setText(String.valueOf(balance));
                JOptionPane.showMessageDialog(this, "Successfully Deposited ₹" + amt + "
New Balance: ₹" + balance, "Success", JOptionPane.INFORMATION_MESSAGE);
            } else if (e.getSource() == btnWithdraw) {
                if (amt <= balance) {
                    balance -= amt;
                    tfBalance.setText(String.valueOf(balance));
                    JOptionPane.showMessageDialog(this, "Successfully Withdrawn ₹" + amt + "
Remaining Balance: ₹" + balance, "Success", JOptionPane.INFORMATION_MESSAGE);
                } else {
                    JOptionPane.showMessageDialog(this, "Insufficient Balance! Current Balance is ₹" + balance, "Error", JOptionPane.ERROR_MESSAGE);
                }
            }
        } catch (NumberFormatException ex) {
            JOptionPane.showMessageDialog(this, "Please enter a valid numeric amount.", "Error", JOptionPane.ERROR_MESSAGE);
        }
    }
